\section{Введение}

\subsection{Контекст проекта}
Крестики-нолики - классическая игра для двух игроков. В данном проекте мы работаем с расширенной версией, включающей:
\begin{itemize}
    \item Игровое поле \textbf{20×20}
    \item \textbf{5 символов в ряд} для победы
    \item \textbf{Препятствия (\#)} на поле
    \item Параметры: \texttt{playable\_part = 0.75}, \texttt{gap = 1}, \texttt{max\_obstacle\_len = 50}
\end{itemize}

\subsection{Цель проекта}
Цель - разработать \textbf{интеллектуального бота}, способного играть в эту игру, соблюдая ограничения по времени (50-100 мс на ход) и достигая удовлетворительных результатов против базового бота (baseline).

\subsection{Структура отчета}
Данный отчет содержит: математическое описание алгоритма, детали реализации на C++, результаты тестирования и заключение.

\section{Математическое описание}

\subsection{Формализация игры}
Пусть \(P\) - игровое поле размера \(M \times N\), где \(M = N = 20\).

Каждая клетка \(p(i,j)\) может содержать:
\begin{itemize}
    \item \textbf{X} (Игрок 1 / наш бот)
    \item \textbf{O} (Игрок 2 / противник)
    \item \textbf{\#} (препятствие)
    \item \textbf{пусто} (свободная клетка)
\end{itemize}
Выигрышная длина: \(L = 5\).

\subsection{Направления поиска}
Пусть \(D\) - множество из 4 направлений:
\[
D = \{ (1,0), (0,1), (1,1), (1,-1) \}
\]

\begin{table}[H]
\centering
\begin{tabular}{|l|l|l|}
\hline
\textbf{Направление} & \textbf{Вектор} & \textbf{Значение} \\
\hline
\(d_1\) & (1,0) & Горизонталь (вправо) \\
\(d_2\) & (0,1) & Вертикаль (вниз) \\
\(d_3\) & (1,1) & Главная диагональ \\
\(d_4\) & (1,-1) & Побочная диагональ \\
\hline
\end{tabular}
\end{table}

\subsection{Функция подсчета}
Для клетки \((x,y)\) и направления \(d = (dx, dy)\):
\[
\text{count}(s, d) = \text{количество символов } s \text{, выстроенных в направлении } d
\]

Подсчет останавливается, когда:
\begin{itemize}
    \item Встречается препятствие \#
    \item Встречается пустая клетка
    \item Достигнута граница поля
\end{itemize}

\subsection{Эвристическая функция оценки}
Пусть \(h(n)\) - функция оценки для линии длины \(n\):

\begin{table}[H]
\centering
\begin{tabular}{|l|l|l|}
\hline
\textbf{n (длина)} & \textbf{h(n)} & \textbf{Значение} \\
\hline
\(n \ge 5\) & 1 000 000 & Немедленная победа \\
\(n = 4\) & 100 000 & Очень опасно \\
\(n = 3\) & 5 000 & Опасно \\
\(n = 2\) & 100 & Небольшое преимущество \\
\(n = 1\) & 1 & Слабо \\
\hline
\end{tabular}
\end{table}

\subsection{Оценка атаки}
Для пустой клетки \((x,y)\):
\[
A(x,y) = \sum_{d \in D} h(\text{count}(\text{мой\_знак}, d))
\]

\subsection{Оценка защиты}
\[
D(x,y) = \sum_{d \in D} h(\text{count}(\text{знак\_противника}, d))
\]

\subsection{Бонус центра}
Центр поля находится в \((10,10)\) (приблизительно):
\[
B(x,y) = 20 - (|x - 10| + |y - 10|)
\]

\begin{table}[H]
\centering
\begin{tabular}{|l|l|l|}
\hline
\textbf{Клетка} & \textbf{Расстояние} & \textbf{Бонус} \\
\hline
(10,10) & 0 & 20 \\
(10,9) & 1 & 19 \\
(5,5) & 10 & 10 \\
(0,0) & 20 & 0 \\
\hline
\end{tabular}
\end{table}

\subsection{Итоговая оценка}
\[
S(x,y) = A(x,y) + 2 \times D(x,y) + B(x,y)
\]

Коэффициент \(\times 2\) для защиты приоритизирует \textbf{блокировку} противника.

\subsection{Выбор хода}
Бот играет клетку, максимизирующую \(S(x,y)\):
\[
\text{оптимальный\_ход} = \arg\max_{x,y | p(x,y)=\text{пусто}} S(x,y)
\]

\subsection{Обработка препятствий}
Препятствия \# обрабатываются как \textbf{стены}:
\begin{itemize}
    \item Бот никогда не может играть на клетке с препятствием
    \item Подсчет выравниваний останавливается на препятствии
\end{itemize}

\subsection{Анализ сложности}
\[
\text{Сложность} = (M \times N) \times |D| \times L = 20 \times 20 \times 4 \times 5 = 8 000 \text{ операций на ход}
\]

\begin{table}[H]
\centering
\begin{tabular}{|l|l|l|}
\hline
\textbf{Параметр} & \textbf{Значение} & \textbf{Значение} \\
\hline
M & 20 & Количество столбцов \\
N & 20 & Количество строк \\
\(|D|\) & 4 & Количество направлений \\
L & 5 & Выигрышная длина \\
\hline
\end{tabular}
\end{table}

\textbf{Пространственная сложность:} \(O(1)\) (без дополнительной памяти)

\subsection{Основные функции}

\subsubsection{Подсчет в направлении}

\begin{lstlisting}
int count_in_direction(const State& state, int x, int y, 
                        int dx, int dy, Sign sign) {
    int count = 1;
    int cols = state.get_opts().cols;
    int rows = state.get_opts().rows;
    
    // Положительное направление
    for (int step = 1; step <= 4; step++) {
        int nx = x + dx * step;
        int ny = y + dy * step;
        if (nx < 0 || nx >= cols || ny < 0 || ny >= rows) break;
        if (state.get_value(nx, ny) == sign) count++;
        else break;
    }
    
    // Отрицательное направление
    for (int step = 1; step <= 4; step++) {
        int nx = x - dx * step;
        int ny = y - dy * step;
        if (nx < 0 || nx >= cols || ny < 0 || ny >= rows) break;
        if (state.get_value(nx, ny) == sign) count++;
        else break;
    }
    return count;
}
\end{lstlisting}

\subsubsection{Оценка клетки}

\begin{lstlisting}
int evaluate_cell(const State& state, int x, int y, Sign sign) {
    int total = 0;
    int dirs[4][2] = {{1,0}, {0,1}, {1,1}, {1,-1}};
    
    for (int i = 0; i < 4; i++) {
        int count = count_in_direction(state, x, y, 
                                        dirs[i][0], dirs[i][1], sign);
        total += score_for_count(count);
    }
    return total;
}
\end{lstlisting}

\subsubsection{Проверка выигрышного хода}

\begin{lstlisting}
bool is_winning_move(const State& state, int x, int y, Sign sign) {
    int dirs[4][2] = {{1,0}, {0,1}, {1,1}, {1,-1}};
    int cols = state.get_opts().cols;
    int rows = state.get_opts().rows;
    
    for (int d = 0; d < 4; d++) {
        int count = 1;
        // Положительное направление
        for (int step = 1; step <= 4; step++) {
            int nx = x + dirs[d][0] * step;
            int ny = y + dirs[d][1] * step;
            if (nx < 0 || nx >= cols || ny < 0 || ny >= rows) break;
            if (state.get_value(nx, ny) == sign) count++;
            else break;
        }
        // Отрицательное направление
        for (int step = 1; step <= 4; step++) {
            int nx = x - dirs[d][0] * step;
            int ny = y - dirs[d][1] * step;
            if (nx < 0 || nx >= cols || ny < 0 || ny >= rows) break;
            if (state.get_value(nx, ny) == sign) count++;
            else break;
        }
        if (count >= 5) return true;
    }
    return false;
}
\end{lstlisting}

\subsubsection{Функция make\_move (основная)}

\begin{lstlisting}
Point MyPlayer::make_move(const State& state) {
    Point best_move = {0, 0};
    int best_score = INT_MIN;
    int cols = state.get_opts().cols;
    int rows = state.get_opts().rows;
    Sign opponent = (m_sign == Sign::X) ? Sign::O : Sign::X;
    
    // ПРИОРИТЕТ 1: Немедленная победа
    for (int x = 0; x < cols; x++) {
        for (int y = 0; y < rows; y++) {
            if (state.get_value(x, y) == Sign::NONE) {
                if (is_winning_move(state, x, y, m_sign)) {
                    return {x, y};
                }
            }
        }
    }
    
    // ПРИОРИТЕТ 2: Блокировка противника
    for (int x = 0; x < cols; x++) {
        for (int y = 0; y < rows; y++) {
            if (state.get_value(x, y) == Sign::NONE) {
                if (is_winning_move(state, x, y, opponent)) {
                    return {x, y};
                }
            }
        }
    }
    
    // ПРИОРИТЕТ 3: Эвристическая оценка
    for (int x = 0; x < cols; x++) {
        for (int y = 0; y < rows; y++) {
            if (state.get_value(x, y) == Sign::NONE) {
                int attack = evaluate_cell(state, x, y, m_sign);
                int defense = evaluate_cell(state, x, y, opponent);
                int center_x = cols / 2;
                int center_y = rows / 2;
                int bonus = 20 - (abs(x - center_x) + abs(y - center_y));
                int score = (attack * 4) + (defense * 2) + bonus;
                if (score > best_score) {
                    best_score = score;
                    best_move = {x, y};
                }
            }
        }
    }
    return best_move;
}
\end{lstlisting}

#include <climits>
#include <cmath>
#include <cstdlib>

namespace ttt::my_player {

const char* MyPlayer::get_name() const {
    return m_name;
}

void MyPlayer::set_sign(game::Sign sign) {
    m_sign = sign;
}

int count_in_direction(const game::State& state, int x, int y, 
                        int dx, int dy, game::Sign sign) {
    int count = 1;
    int cols = state.get_opts().cols;
    int rows = state.get_opts().rows;
    
    for (int step = 1; step <= 4; step++) {
        int nx = x + dx * step;
        int ny = y + dy * step;
        if (nx < 0 || nx >= cols || ny < 0 || ny >= rows) break;
        if (state.get_value(nx, ny) == sign) count++;
        else break;
    }
    
    for (int step = 1; step <= 4; step++) {
        int nx = x - dx * step;
        int ny = y - dy * step;
        if (nx < 0 || nx >= cols || ny < 0 || ny >= rows) break;
        if (state.get_value(nx, ny) == sign) count++;
        else break;
    }
    return count;
}

int score_for_count(int count) {
    if (count >= 5) return 1000000;
    if (count == 4) return 100000;
    if (count == 3) return 5000;
    if (count == 2) return 100;
    return 1;
}

int evaluate_cell(const game::State& state, int x, int y, game::Sign sign) {
    int total = 0;
    int dirs[4][2] = {{1,0}, {0,1}, {1,1}, {1,-1}};
    
    for (int i = 0; i < 4; i++) {
        int count = count_in_direction(state, x, y, 
                                        dirs[i][0], dirs[i][1], sign);
        total += score_for_count(count);
    }
    return total;
}

bool is_winning_move(const game::State& state, int x, int y, game::Sign sign) {
    int dirs[4][2] = {{1,0}, {0,1}, {1,1}, {1,-1}};
    int cols = state.get_opts().cols;
    int rows = state.get_opts().rows;
    
    for (int d = 0; d < 4; d++) {
        int count = 1;
        for (int step = 1; step <= 4; step++) {
            int nx = x + dirs[d][0] * step;
            int ny = y + dirs[d][1] * step;
            if (nx < 0 || nx >= cols || ny < 0 || ny >= rows) break;
            if (state.get_value(nx, ny) == sign) count++;
            else break;
        }
        for (int step = 1; step <= 4; step++) {
            int nx = x - dirs[d][0] * step;
            int ny = y - dirs[d][1] * step;
            if (nx < 0 || nx >= cols || ny < 0 || ny >= rows) break;
            if (state.get_value(nx, ny) == sign) count++;
            else break;
        }
        if (count >= 5) return true;
    }
    return false;
}

game::Point MyPlayer::make_move(const game::State& state) {
    game::Point best_move = {0, 0};
    int best_score = INT_MIN;
    int cols = state.get_opts().cols;
    int rows = state.get_opts().rows;
    game::Sign opponent = (m_sign == game::Sign::X) ? game::Sign::O : game::Sign::X;
    
    // ПРИОРИТЕТ 1: Немедленная победа
    for (int x = 0; x < cols; x++) {
        for (int y = 0; y < rows; y++) {
            if (state.get_value(x, y) == game::Sign::NONE) {
                if (is_winning_move(state, x, y, m_sign)) {
                    return {x, y};
                }
            }
        }
    }
    
    // ПРИОРИТЕТ 2: Блокировка противника
    for (int x = 0; x < cols; x++) {
        for (int y = 0; y < rows; y++) {
            if (state.get_value(x, y) == game::Sign::NONE) {
                if (is_winning_move(state, x, y, opponent)) {
                    return {x, y};
                }
            }
        }
    }
    
    // ПРИОРИТЕТ 3: Эвристическая оценка
    for (int x = 0; x < cols; x++) {
        for (int y = 0; y < rows; y++) {
            if (state.get_value(x, y) == game::Sign::NONE) {
                int attack = evaluate_cell(state, x, y, m_sign);
                int defense = evaluate_cell(state, x, y, opponent);
                int center_x = cols / 2;
                int center_y = rows / 2;
                int bonus = 20 - (abs(x - center_x) + abs(y - center_y));
                int score = (attack * 4) + (defense * 2) + bonus;
                if (score > best_score) {
                    best_score = score;
                    best_move = {x, y};
                }
            }
        }
    }
    return best_move;
}
